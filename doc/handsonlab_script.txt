    - Fuseki UI에 접속해서 아래 SPARQL QUERY를 실행 시킴. 아래 위치에 sql 있음.

    vi query.sparql
    
    ============================================================================

    PREFIX :    <http://k.doverenc/onto#>
    PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

    SELECT ?t ?level ?min ?max
        ?ratioPerCapita ?minStaff
        ?hour
        ?securityMul
        ?female ?age20 ?age30
        ?staffBase ?staffAfterMul ?staffFinal
        ?secStaff ?guideStaff ?cashierStaff
    WHERE {
    # 1) 예측 (강남역)
    GRAPH <http://k.doverenc/onto> {
        ?f a :Forecast ;
        :forPlace :place_POI014 ;
        :forecastTime ?t ;
        :forecastCongestionLevel ?level ;
        :forecastPopulationMin ?min ;
        :forecastPopulationMax ?max .
    }

    # 2) 혼잡도 → 정책 매핑
    GRAPH <http://k.doverenc/onto> {
        ?policy a :StaffPolicy ;
                :appliesLevel ?level ;
                :ratioPerCapita ?ratioPerCapita ;
                :minStaff ?minStaff ;
                :roleWeightSecurity ?wSec ;
                :roleWeightGuide    ?wGuide ;
                :roleWeightCashier  ?wCash .
    }

    # 3) 장소 인구통계 (보정 규칙용)
    GRAPH <http://k.doverenc/onto> {
        :place_POI014 :femaleRate ?female ;
                    :ageRate_20 ?age20 ;
                    :ageRate_30 ?age30 .
    }

    # 4) 시간대 가중치 (옵션)
    BIND(HOURS(?t) AS ?hour)
    OPTIONAL {
        GRAPH <http://k.doverenc/onto> {
        ?tb a :TimeBand ;
            :bandStartHour ?hS ; :bandEndHour ?hE ;
            :securityMultiplier ?securityMul0 .
        }
        FILTER(?hS <= ?hour && ?hour < ?hE)
    }
    BIND(COALESCE(?securityMul0, 1.0) AS ?securityMul)

    # 5) 기본 인원 (비율/최소)
    BIND( CEIL( xsd:decimal(?max) / xsd:decimal(?ratioPerCapita) ) AS ?staffByRatio )
    BIND( IF(?staffByRatio < ?minStaff, ?minStaff, ?staffByRatio) AS ?staffBase )

    # 6) 시간대 가중치 적용 (총원은 보전, 분배 비중만 보정)
    BIND( ?wSec * ?securityMul AS ?wSec2 )
    BIND( (?wSec2 + ?wGuide + ?wCash) AS ?wSum )
    BIND( ?staffBase AS ?staffAfterMul )

    # 7) 인구통계 보정 (예: 여성비 > threshold면 가이드 +extra)
    OPTIONAL {
        GRAPH <http://k.doverenc/onto> {
        ?rule a :DemographicRule ;
                :metric :femaleRate ;
                :threshold ?th ;
                :extraStaff ?extra ;
                :extraStaffRole :RoleGuide .
        }
    }
    BIND( IF(BOUND(?rule) && xsd:decimal(?female) > xsd:decimal(?th),
            xsd:decimal(?extra), 0) AS ?demoExtra )

    # 8) 역할 배분 (정규화 후 배분)
    BIND( CEIL(?staffAfterMul * (?wSec2 / ?wSum)) AS ?secStaff0 )
    BIND( CEIL(?staffAfterMul * (?wGuide  / ?wSum)) AS ?guideStaff0 )
    BIND( ?staffAfterMul - ?secStaff0 - ?guideStaff0 AS ?cashierStaff0 )

    # 9) 최종 (인구통계 보정 반영)
    BIND( ?secStaff0 AS ?secStaff )
    BIND( ?guideStaff0 + ?demoExtra AS ?guideStaff )
    BIND( ?cashierStaff0 AS ?cashierStaff )

    # 10) 총원 산출
    BIND( (?secStaff + ?guideStaff + ?cashierStaff) AS ?staffFinal )
    }
    ORDER BY ?t

    ============================================================================

1. JSON, TTL , FOL (First Order Logic)

- JSON : 원천 데이터
- TTL : Ontology가 이해할 수 있는 데이터. RDF를 구현하는 표현식
- FOL : 복잡한 데이터를 논리적으로 표현하는 언어. 주어/서술어/목적어 로 구성.

  JSON (Data) -> Ontology (지식) -> FOL - 논리 표현 (추론)
  
  
2. 서울시 실시간 데이터를 FOL로 표현하기

- JSON : KEY / VALUE 표현 일 뿐, 의미적 맥락이 없어서 기계가 자동으로 판단 할 수 없다.
- RDF/TTL : 데이터에 의미를 부여

- Ontology : 무엇이 존재하는가?에 답한다. 세상에 어떤 개념들이 존재하고 그들 간의 관계는 어떠한지를 정의
- FOL : 무엇이 사실인가?에 답한다. 정의된 개념들에 대한 구체적인 사실과 규칙을 논리적으로 표현

3. 전체 여정 요약

  JSON 데이터            -> RDF/TTL         -> 온톨로지       -> FOL 규칙      -> 지식 추론
  구조화된 원본 데이터     트리플 구조 변환     개념 구조 정의     논리 규칙 추가      의미적 질의 및 통찰

  "데이터는 과거를 설명하지만, 지식은 미래를 예측한다."

    # GeoSPARQL 검증

    - Fuseki UI에서 아래 SPARQL 실행 - Fuseki에서 GeoSPARQL 정상 동작하는지 확인 목적
      (** Fuseki의 Geo 계산 방식이 완벽하지 않아 표준을 지원하지 못함)

    ============================================================================

    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX spatialf: <http://jena.apache.org/function/spatial#>
    PREFIX uom: <http://www.opengis.net/def/uom/OGC/1.0/>

    SELECT ?distance
    WHERE {
      BIND("POINT(127.05 37.50)"^^geo:wktLiteral AS ?p1)
      BIND("POINT(127.06 37.51)"^^geo:wktLiteral AS ?p2)
      BIND(spatialf:distance(?p1, ?p2, uom:metre) AS ?distance)
    }
    
    ============================================================================

    -> 1419.349865379146e0 값이 출력되어야 한다.

    ============================================================================

    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX spatialf: <http://jena.apache.org/function/spatial#>
    PREFIX uom: <http://www.opengis.net/def/uom/OGC/1.0/>
    PREFIX cctv: <http://k.doverenc/onto/cctv#>

    SELECT ?cctv ?name ?wkt
    WHERE {
      BIND("<http://www.opengis.net/def/crs/EPSG/0/4326> POINT(127.0276 37.4979)"^^geo:wktLiteral AS ?center)

      ?cctv a cctv:CCTV ;
        rdfs:label ?name ;
        geo:hasGeometry [ geo:asWKT ?wkt ] .  
      FILTER(spatialf:distance(?wkt, ?center, uom:metre) < 500 )
    }

    ============================================================================

    ============================================================================

    PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
    PREFIX geo: <http://www.opengis.net/ont/geosparql#>
    PREFIX spatialf: <http://jena.apache.org/function/spatial#>
    PREFIX uom: <http://www.opengis.net/def/uom/OGC/1.0/>
    PREFIX cctv: <http://k.doverenc/onto/cctv#>
    SELECT ?cctv?name ?url
    WHERE {
      BIND( "<http://www.opengis.net/def/crs/EPSG/0/4326> POINT(127.0276 37.4979)"^^geo:wktLiteral AS ?center )
      BIND( 1000 AS ?radius ) # 미터
  
      ?cctv a cctv:CCTV;
        rdfs:label?name ;
        geo:hasGeometry[ geo:asWKT?wkt] ;
        cctv:hasStreamUrl?url.

      FILTER( spatialf:distance(?wkt, ?center, uom:metre) < ?radius )
    }

    ============================================================================

Types: deb
URIs: https://download.docker.com/linux/ubuntu
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: stable
Architectures: $(dpkg --print-architecture)
Signed-By: /etc/apt/keyrings/docker.asc
EOF

- 모든 docker volumn 제거
sudo docker volume prune -f

- 모든 docker data 제거
sudo docker system prune -af


# docker container 자동 재기동
# - OS 재기동 등으로 DOCKER가 종료될 경우 docker container가 자동으로 실행되지 않는다.
# - 아래 명령을 이용하면 자동 재실행이 된다.

docker update --restart=always <container_id>
